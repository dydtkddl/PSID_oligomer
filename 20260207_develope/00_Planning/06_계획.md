def calculate_van_hove(trajectory, times=[1, 10, 50, 100]):
    """
    G_s(r,t) = <δ(r - |r_i(t) - r_i(0)|)>
    Discrete peaks → Hopping mechanism
    Gaussian → Continuous diffusion
    """
    for t_ps in times:
        # 시간 지연 t에서의 변위 분포 계산
        displacements = calculate_displacements(trajectory, t_ps)
        hist, bins = np.histogram(displacements, bins=100, density=True)

        # 피크 검출
        peaks = find_peaks(hist, prominence=0.01)
        if len(peaks) > 1:
            print(f"t={t_ps}ps: {len(peaks)} discrete peaks → HOPPING!")
```

def calculate_anisotropic_diffusion(trajectory):
    """
    D_parallel vs D_perpendicular 비교
    D_∥ >> D_⊥ 이면 directed transport 증거
    """
    # 백본 방향 vs 수직 방향 MSD 분해
    D_parallel = fit_msd_parallel(trajectory)
    D_perp = fit_msd_perpendicular(trajectory)

    anisotropy_ratio = D_parallel / D_perp
    return anisotropy_ratio
```

**전기적 성질 예측**

```python
def calculate_ionic_conductivity(trajectory, T=300):
    """
    Nernst-Einstein: σ = (e²/k_BT) * n * q² * D
    """
    D_tfsi = calculate_diffusion_coefficient(trajectory)
    n_ions = calculate_number_density(trajectory)

    e = 1.602e-19  # C
    k_B = 1.381e-23  # J/K

    sigma = (e**2 / (k_B * T)) * n_ions * D_tfsi * 1e-4  # S/cm
    return sigma

# 온도별 Arrhenius 분석
def arrhenius_analysis(temperatures, conductivities):
    """ln(σ) = ln(σ₀) - E_a/(k_BT)"""
    x = 1000 / np.array(temperatures)
    y = np.log(conductivities)
    slope, intercept = np.polyfit(x, y, 1)
    E_a = -slope * 8.617e-5 * 1000  # eV
    return E_a
```
